# Part 3: Adding Elements to Arrays

# Task 3.1: Add Single Element ($push)
db.products.updateOne(
  { sku: "CAMERA-CANON-EOS-R5" },
  { $push: { tags: "featured" } }
);

{
  acknowledged: true,
  insertedId: null,
  matchedCount: 1,
  modifiedCount: 1,
  upsertedCount: 0
}

# Task 3.2: Add Multiple Elements
db.products.updateOne(
  { sku: "CAMERA-CANON-EOS-R5" },
  { $push: { tags: { $each: ["new-arrival", "bestseller"] } } }
);

{
  acknowledged: true,
